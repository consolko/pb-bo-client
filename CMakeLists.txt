cmake_minimum_required(VERSION 3.21)
project(bookorbit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BOOKORBIT_VERSION "1.1.0" CACHE STRING "Application version")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/resources/update-public-key.hex" BOOKORBIT_UPDATE_PUBLIC_KEY)
string(STRIP "${BOOKORBIT_UPDATE_PUBLIC_KEY}" BOOKORBIT_UPDATE_PUBLIC_KEY)
string(LENGTH "${BOOKORBIT_UPDATE_PUBLIC_KEY}" BO_KEY_LENGTH)
if(NOT BO_KEY_LENGTH EQUAL 64 OR NOT BOOKORBIT_UPDATE_PUBLIC_KEY MATCHES "^[a-f0-9]+$")
  message(FATAL_ERROR "The release public key must contain 32 bytes in lowercase hex")
endif()
add_compile_definitions(BOOKORBIT_VERSION="${BOOKORBIT_VERSION}" BOOKORBIT_UPDATE_PUBLIC_KEY="${BOOKORBIT_UPDATE_PUBLIC_KEY}")
set(SOURCES src/i18n.h src/main.cpp src/client.cpp src/client.h src/progress.cpp src/progress.h src/device.cpp src/device.h src/ota.cpp src/ota.h src/update.cpp src/update.h)
if(POCKETBOOK_TOOLCHAIN_LOADED)
  if(NOT POCKETBOOK_QT_SDK)
    message(FATAL_ERROR "Set POCKETBOOK_QT_SDK to the pinned pocketbook-sdk-qt6 checkout")
  endif()
  set(POCKETBOOK_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
  add_subdirectory("${POCKETBOOK_QT_SDK}" "${CMAKE_BINARY_DIR}/sdk")
  pocketbook_add_app(bookorbit SOURCES ${SOURCES} QRC qml/app.qrc LIBRARIES Qt6::Network Qt6::Sql Qt6::Xml z)
  target_compile_definitions(bookorbit PRIVATE POCKETBOOK_DEVICE)
else()
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick Network Xml)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_executable(bookorbit ${SOURCES} qml/app.qrc)
  target_link_libraries(bookorbit PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network Qt6::Xml z)
endif()
target_compile_options(bookorbit PRIVATE -Wall -Wextra -Wpedantic)

  add_test(NAME client_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/run_desktop_checks.py $<TARGET_FILE:client_check> --native-layout)
  add_test(NAME audit_regressions COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/run_desktop_checks.py $<TARGET_FILE:audit_check>)
  add_executable(ota_check tools/ota_check.cpp src/device.cpp src/device.h src/update.cpp src/update.h src/ota.cpp src/client.cpp src/client.h src/progress.cpp src/progress.h qml/app.qrc)
  target_link_libraries(ota_check PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network Qt6::Xml z)
  target_include_directories(ota_check PRIVATE src)
