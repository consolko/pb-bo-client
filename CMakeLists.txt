cmake_minimum_required(VERSION 3.21)
project(bookorbit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOURCES src/main.cpp src/client.cpp src/client.h src/device.cpp src/device.h)
if(POCKETBOOK_TOOLCHAIN_LOADED)
  if(NOT POCKETBOOK_QT_SDK)
    message(FATAL_ERROR "Set POCKETBOOK_QT_SDK to the pinned pocketbook-sdk-qt6 checkout")
  endif()
  set(POCKETBOOK_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
  add_subdirectory("${POCKETBOOK_QT_SDK}" "${CMAKE_BINARY_DIR}/sdk")
  pocketbook_add_app(bookorbit SOURCES ${SOURCES} QRC qml/app.qrc LIBRARIES Qt6::Network)
  target_compile_definitions(bookorbit PRIVATE POCKETBOOK_DEVICE)
else()
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick Network)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  add_executable(bookorbit ${SOURCES} qml/app.qrc)
  target_link_libraries(bookorbit PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network)
endif()
target_compile_options(bookorbit PRIVATE -Wall -Wextra -Wpedantic)

option(BOOKORBIT_CHECKS "Build local client integration check" OFF)
if(BOOKORBIT_CHECKS)
  if(POCKETBOOK_TOOLCHAIN_LOADED)
    pocketbook_add_app(client_check SOURCES tools/client_check.cpp src/client.cpp src/client.h LIBRARIES Qt6::Network)
  else()
    add_executable(client_check tools/client_check.cpp src/client.cpp src/client.h)
    target_link_libraries(client_check PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
  endif()
  target_include_directories(client_check PRIVATE src)
  target_compile_options(client_check PRIVATE -Wall -Wextra -Wpedantic)
  if(POCKETBOOK_TOOLCHAIN_LOADED)
    pocketbook_add_app(reader_state_check SOURCES tools/reader_state_check.cpp src/device.cpp src/device.h)
    target_compile_definitions(reader_state_check PRIVATE POCKETBOOK_DEVICE)
    target_include_directories(reader_state_check PRIVATE src)
  endif()
endif()
